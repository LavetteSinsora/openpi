#
#   pip install -e packages/openpi-client
#   pip install -r ego2g1/deploy/requirements.txt
#

#   unitree_sdk2py  — git clone https://github.com/unitreerobotics/unitree_sdk2_python

#   unitree_deploy  — the vendor repo, for its ImageClient camera only
#
# And on the G1 board itself, the image server must be running or the camera
# never produces a frame:
#   ssh unitree@192.168.123.164   # password: 123
#   conda activate tv && cd ~/image_server && python image_server.py
